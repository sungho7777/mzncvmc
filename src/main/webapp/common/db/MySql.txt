# 테이블 생성
CREATE TABLE company (
    company_id BIGINT AUTO_INCREMENT PRIMARY KEY COMMENT '회사 고유 ID',
    company_name VARCHAR(200) NOT NULL COMMENT '회사명',
    company_eng_name VARCHAR(200) DEFAULT NULL COMMENT '영문 회사명',
    business_number VARCHAR(20) UNIQUE NOT NULL COMMENT '사업자 등록번호',
    ceo_name VARCHAR(100) DEFAULT NULL COMMENT '대표자명',
    established_date DATE DEFAULT NULL COMMENT '설립일',
    company_type VARCHAR(50) DEFAULT NULL COMMENT '회사 형태(예: 주식회사, LLC 등)',
    industry VARCHAR(100) DEFAULT NULL COMMENT '업종',
    phone VARCHAR(20) DEFAULT NULL COMMENT '대표 전화번호',
    fax VARCHAR(20) DEFAULT NULL COMMENT '팩스 번호',
    email VARCHAR(100) DEFAULT NULL COMMENT '대표 이메일',
    website VARCHAR(200) DEFAULT NULL COMMENT '홈페이지',

    postal_code VARCHAR(10) DEFAULT NULL COMMENT '우편번호',
    address VARCHAR(300) DEFAULT NULL COMMENT '주소',
    address_detail VARCHAR(300) DEFAULT NULL COMMENT '상세 주소',

    created_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP COMMENT '등록일시',
    update_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '수정일시',

    status ENUM('ACTIVE','INACTIVE') DEFAULT 'ACTIVE' COMMENT '상태'
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='회사 정보 테이블';

# 테이블 생성
CREATE TABLE users (
        user_id INT AUTO_INCREMENT PRIMARY KEY COMMENT '사용자 고유 ID',
        company_id BIGINT NOT NULL COMMENT '소속 회사 ID (FK: company.company_id)',
        username VARCHAR(100) NOT NULL UNIQUE COMMENT '로그인 아이디',
        password VARCHAR(255) NOT NULL COMMENT '암호화된 비밀번호',
        full_name VARCHAR(100) DEFAULT NULL COMMENT '사용자 이름',
        email VARCHAR(100) DEFAULT NULL UNIQUE COMMENT '이메일',
        provider ENUM('LOCAL', 'GOOGLE', 'KAKAO', 'NAVER', 'GIT') DEFAULT 'LOCAL' COMMENT 'OAUTH2 provider',
        provider_id VARCHAR(20) DEFAULT NULL COMMENT 'OAUTH2 provider_id',
        phone VARCHAR(20) DEFAULT NULL COMMENT '전화번호',
        role ENUM('ADMIN','MANAGER','USER') DEFAULT 'USER' COMMENT '권한',
        status ENUM('ACTIVE','INACTIVE') DEFAULT 'ACTIVE' COMMENT '계정 상태',
        connected ENUM('N','Y') DEFAULT 'N' COMMENT '접속여부',
        pw_notify_duration VARCHAR(3) DEFAULT '999' COMMENT '비밀번호 알림기간',

        created_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP COMMENT '등록일시',
        update_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '수정일시',
        last_login TIMESTAMP NULL DEFAULT NULL COMMENT '마지막 로그인 시각',

        CONSTRAINT fk_company FOREIGN KEY (company_id) REFERENCES company(company_id)
            ON DELETE CASCADE
            ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='사용자 정보 테이블';

# 펌부파일 테이블 생성
CREATE TABLE files (
    file_id BIGINT AUTO_INCREMENT PRIMARY KEY,
    reference_type VARCHAR(50) NOT NULL COMMENT '참조 테이블명 (user, company, bbs 등)',
    reference_id BIGINT NOT NULL COMMENT '참조 테이블의 ID',
    original_filename VARCHAR(255) NOT NULL COMMENT '원본 파일명',
    stored_filename VARCHAR(255) NOT NULL COMMENT '저장된 파일명 (UUID)',
    file_path VARCHAR(500) NOT NULL COMMENT '파일 저장 경로',
    file_size BIGINT NOT NULL COMMENT '파일 크기 (bytes)',
    file_extension VARCHAR(10) NOT NULL COMMENT '파일 확장자',
    mime_type VARCHAR(100) COMMENT 'MIME 타입',
    upload_date DATETIME DEFAULT CURRENT_TIMESTAMP COMMENT '업로드 일시',
    created_by BIGINT COMMENT '업로드한 사용자 ID',
    is_deleted BOOLEAN DEFAULT FALSE COMMENT '삭제 여부',
    deleted_date DATETIME COMMENT '삭제 일시',

    INDEX idx_reference (reference_type, reference_id),
    INDEX idx_created_by (created_by),
    INDEX idx_upload_date (upload_date)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='첨부파일 관리 테이블';


# user_history 테이블 생성
CREATE TABLE user_history (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    username VARCHAR(100) NOT NULL COMMENT '사용자 ID',
    action_type VARCHAR(20) NOT NULL COMMENT '액션 타입 (LOGIN, LOGOUT, CREATE, READ, UPDATE, DELETE)',
    uri VARCHAR(500) NOT NULL COMMENT '요청 URI',
    http_method VARCHAR(10) NOT NULL COMMENT 'HTTP 메소드 (GET, POST, PUT, DELETE)',
    controller_method VARCHAR(100) COMMENT '컨트롤러 메소드명 (예: UserController.getUser)',
    request_data TEXT COMMENT '요청 파라미터 또는 중요 데이터',
    client_ip VARCHAR(50) COMMENT '클라이언트 IP 주소',
    created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '생성 시간'
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='사용자 활동 히스토리';

# 인덱스 생성 (성능 최적화)
CREATE INDEX idx_user_time ON user_history(username, created_at);
CREATE INDEX idx_action_type ON user_history(action_type);
CREATE INDEX idx_created_at ON user_history(created_at);


# 로그인 2단계 인증(OTP) 관리 테이블
DROP TABLE IF EXISTS user_otp;
CREATE TABLE user_otp (
    id BIGINT AUTO_INCREMENT PRIMARY KEY COMMENT 'OTP 이력 고유 ID',
    user_id BIGINT NOT NULL COMMENT '사용자 ID (users.user_id)',
    otp_code VARCHAR(6) NOT NULL COMMENT '로그인용 1회성 인증 코드',
    expired_at DATETIME NOT NULL COMMENT 'OTP 만료 시각',
    used_yn CHAR(1) DEFAULT 'N' COMMENT 'OTP 사용 여부 (Y: 사용됨, N: 미사용)',
    fail_count INT DEFAULT 0 COMMENT 'OTP 인증 실패 횟수',
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP COMMENT 'OTP 생성 시각',
    INDEX idx_user_id (user_id) COMMENT '사용자별 OTP 조회용 인덱스'
) ENGINE=InnoDB
  DEFAULT CHARSET=utf8mb4
  COMMENT='로그인 2단계 인증(OTP) 관리 테이블';

# 사용자 MFA(다중 인증) 정보 테이블
DROP TABLE IF EXISTS user_mfa;
CREATE TABLE user_mfa (
    id BIGINT NOT NULL AUTO_INCREMENT COMMENT 'MFA 테이블 PK',
    user_id BIGINT NOT NULL COMMENT '사용자 ID (users.id 참조)',
    mfa_enabled BOOLEAN NOT NULL DEFAULT FALSE COMMENT 'MFA 활성화 여부',
    mfa_secret TEXT COMMENT 'MFA 비밀키 (AES 암호화 저장)',
    mfa_verified BOOLEAN NOT NULL DEFAULT FALSE COMMENT 'MFA 최초 인증 완료 여부',
    backup_codes TEXT COMMENT 'MFA 백업 코드 (JSON 또는 암호화 저장)',
    failed_attempts BIGINT NOT NULL COMMENT '실패 횟수',
    locked_until DATETIME COMMENT '잠금 해제 시간',
    created_at DATETIME COMMENT '생성 일시',
    updated_at DATETIME COMMENT '수정 일시',
    last_verified_at DATETIME COMMENT '마지막 MFA 인증 일시',
    PRIMARY KEY (id),
    UNIQUE KEY uk_user_mfa_user_id (user_id)
) ENGINE=InnoDB
  DEFAULT CHARSET=utf8mb4
  COLLATE=utf8mb4_unicode_ci
  COMMENT='사용자 MFA(다중 인증) 정보 테이블';

SELECT * FROM user_mfa;

# DCS Log 분석용 테이블
DROP TABLE IF EXISTS dcs_log;
CREATE TABLE dcs_log (
    log_id    BIGINT NOT NULL AUTO_INCREMENT,
    server_no VARCHAR(255) NULL COMMENT '',
    log_date  VARCHAR(255) NULL COMMENT '',
    log_time  VARCHAR(255) NULL COMMENT '',
    random_no VARCHAR(255) NULL COMMENT '',
    user_id   VARCHAR(255) NULL COMMENT '',
    hsb       VARCHAR(255) NULL COMMENT '',
    screen_id VARCHAR(255) NULL COMMENT '',
    url       VARCHAR(255) NULL COMMENT '',
    action    VARCHAR(255) NULL COMMENT '',
    ip        VARCHAR(255) NULL COMMENT '',
    type      VARCHAR(255) NULL COMMENT '',
    PRIMARY KEY (log_id)
) ENGINE=InnoDB
  DEFAULT CHARSET=utf8mb4
  COLLATE=utf8mb4_unicode_ci
  COMMENT='DCS Log 분석용 테이블';

SELECT * FROM dcs_log;
