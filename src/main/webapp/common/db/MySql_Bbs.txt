DROP TABLE IF EXISTS bbs_reports;
DROP TABLE IF EXISTS bbs_user_settings;
DROP TABLE IF EXISTS bbs_post_tags;
DROP TABLE IF EXISTS bbs_tags;
DROP TABLE IF EXISTS bbs_view_logs;
DROP TABLE IF EXISTS bbs_votes;

# BBS 카테고리 테이블 (게시판 분류)
CREATE TABLE bbs_categories (
    category_id BIGINT AUTO_INCREMENT PRIMARY KEY,
    category_name VARCHAR(100) NOT NULL COMMENT '카테고리명',
    category_code VARCHAR(50) NOT NULL UNIQUE COMMENT '카테고리 코드',
    description TEXT COMMENT '카테고리 설명',
    sort_order INT DEFAULT 0 COMMENT '정렬 순서',
    is_active BOOLEAN DEFAULT TRUE COMMENT '활성 여부',
    allow_anonymous BOOLEAN DEFAULT FALSE COMMENT '익명 게시 허용',
    allow_file_upload BOOLEAN DEFAULT TRUE COMMENT '파일 업로드 허용',
    max_file_count INT DEFAULT 10 COMMENT '최대 파일 개수',
    read_permission VARCHAR(50) DEFAULT 'ALL' COMMENT '읽기 권한 (ALL, USER, ADMIN)',
    write_permission VARCHAR(50) DEFAULT 'USER' COMMENT '쓰기 권한 (USER, ADMIN)',
    created_date DATETIME DEFAULT CURRENT_TIMESTAMP,
    updated_date DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    created_by BIGINT COMMENT '생성자 ID',

    INDEX idx_category_code (category_code),
    INDEX idx_sort_order (sort_order),
    INDEX idx_is_active (is_active)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='BBS 카테고리';

# BBS 게시글 테이블
CREATE TABLE bbs_posts (
    # 기본정보
    post_id BIGINT AUTO_INCREMENT PRIMARY KEY,
    category_id BIGINT NOT NULL COMMENT '카테고리 ID',
    title VARCHAR(500) NOT NULL COMMENT '제목',
    bbs_content LONGTEXT NOT NULL COMMENT '내용',
    author_id BIGINT COMMENT '작성자 ID (NULL이면 익명)',
    author_name VARCHAR(100) COMMENT '작성자명 (익명 또는 닉네임)',
    author_ip VARCHAR(45) COMMENT '작성자 IP',
    password VARCHAR(100) COMMENT '익명 게시글 비밀번호 (암호화)',

    # 게시글 상태
    status ENUM('ACTIVE', 'HIDDEN', 'DELETED', 'BLOCKED') DEFAULT 'ACTIVE' COMMENT '상태 (ACTIVE, HIDDEN, DELETED, BLOCKED)',

    is_notice BOOLEAN DEFAULT FALSE COMMENT '공지사항 여부',
    is_top_fixed BOOLEAN DEFAULT FALSE COMMENT '상단 고정 여부',
    is_secret BOOLEAN DEFAULT FALSE COMMENT '비밀글 여부',

    # 조회/통계
    view_count BIGINT DEFAULT 0 COMMENT '조회수',
    like_count BIGINT DEFAULT 0 COMMENT '좋아요 수',
    dislike_count BIGINT DEFAULT 0 COMMENT '싫어요 수',
    comment_count BIGINT DEFAULT 0 COMMENT '댓글 수',
    file_count INT DEFAULT 0 COMMENT '첨부파일 수',

    # 답글 관련 (계층형 게시판)
    parent_id BIGINT COMMENT '부모 게시글 ID',
    depth INT DEFAULT 0 COMMENT '답글 깊이',
    group_id BIGINT COMMENT '답글 그룹 ID',
    group_order INT DEFAULT 0 COMMENT '그룹내 순서',

    # 메타 정보
    tags VARCHAR(500) COMMENT '태그 (쉼표 구분)',
    meta_data JSON COMMENT '추가 메타 데이터',

    # 시간 정보
    created_date DATETIME DEFAULT CURRENT_TIMESTAMP COMMENT '생성 일시',
    updated_date DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '수정 일시',
    deleted_date DATETIME COMMENT '삭제 일시',

    FOREIGN KEY (category_id) REFERENCES bbs_categories(category_id),
    FOREIGN KEY (parent_id) REFERENCES bbs_posts(post_id),

    INDEX idx_category_status (category_id, status),
    INDEX idx_author_id (author_id),
    INDEX idx_created_date (created_date DESC),
    INDEX idx_view_count (view_count DESC),
    INDEX idx_is_notice (is_notice, is_top_fixed),
    INDEX idx_parent_group (parent_id, group_id, group_order),
    INDEX idx_status_created (status, created_date DESC),
    FULLTEXT INDEX ft_title_content (title, bbs_content)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='BBS 게시글';

# BBS 댓글 테이블
CREATE TABLE bbs_comments (
    comment_id BIGINT AUTO_INCREMENT PRIMARY KEY,
    post_id BIGINT NOT NULL COMMENT '게시글 ID',
    parent_id BIGINT COMMENT '부모 댓글 ID (대댓글)',
    depth INT DEFAULT 0 COMMENT '댓글 깊이',
    bbs_content TEXT NOT NULL COMMENT '댓글 내용',

    author_id BIGINT COMMENT '작성자 ID',
    author_name VARCHAR(100) COMMENT '작성자명',
    author_ip VARCHAR(45) COMMENT '작성자 IP',
    password VARCHAR(100) COMMENT '익명 댓글 비밀번호',

    status VARCHAR(20) DEFAULT 'ACTIVE' COMMENT '상태 (ACTIVE, HIDDEN, DELETED, BLOCKED)',
    like_count BIGINT DEFAULT 0 COMMENT '좋아요 수',
    dislike_count BIGINT DEFAULT 0 COMMENT '싫어요 수',

    created_date DATETIME DEFAULT CURRENT_TIMESTAMP,
    updated_date DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    deleted_date DATETIME COMMENT '삭제 일시',

    FOREIGN KEY (post_id) REFERENCES bbs_posts(post_id) ON DELETE CASCADE,
    FOREIGN KEY (parent_id) REFERENCES bbs_comments(comment_id),

    INDEX idx_post_id (post_id),
    INDEX idx_parent_depth (parent_id, depth),
    INDEX idx_author_id (author_id),
    INDEX idx_status_created (status, created_date),
    INDEX idx_created_date (created_date)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='BBS 댓글';

# 게시글/댓글 추천/비추천 테이블
CREATE TABLE bbs_votes (
    vote_id BIGINT AUTO_INCREMENT PRIMARY KEY,
    target_type VARCHAR(20) NOT NULL COMMENT '대상 타입 (POST, COMMENT)',
    target_id BIGINT NOT NULL COMMENT '대상 ID',
    user_id BIGINT NOT NULL COMMENT '투표한 사용자 ID',
    vote_type VARCHAR(10) NOT NULL COMMENT '투표 타입 (LIKE, DISLIKE)',
    ip_address VARCHAR(45) COMMENT 'IP 주소',
    created_date DATETIME DEFAULT CURRENT_TIMESTAMP,

    UNIQUE KEY uk_user_target (target_type, target_id, user_id),
    INDEX idx_target (target_type, target_id),
    INDEX idx_user_id (user_id),
    INDEX idx_created_date (created_date)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='게시글/댓글 투표';

# 게시글 조회 기록 테이블
CREATE TABLE bbs_view_logs (
    view_id BIGINT AUTO_INCREMENT PRIMARY KEY,
    post_id BIGINT NOT NULL,
    user_id BIGINT COMMENT '조회한 사용자 ID (NULL이면 비회원)',
    ip_address VARCHAR(45) NOT NULL,
    user_agent TEXT COMMENT '사용자 에이전트',
    referer VARCHAR(500) COMMENT '리퍼러',
    created_date DATETIME DEFAULT CURRENT_TIMESTAMP,

    INDEX idx_post_id (post_id),
    INDEX idx_user_id (user_id),
    INDEX idx_ip_date (ip_address, created_date),
    INDEX idx_created_date (created_date)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='게시글 조회 로그';

# 게시글 태그 테이블 (정규화된 태그 관리)
CREATE TABLE bbs_tags (
    tag_id BIGINT AUTO_INCREMENT PRIMARY KEY,
    tag_name VARCHAR(50) NOT NULL UNIQUE COMMENT '태그명',
    use_count BIGINT DEFAULT 0 COMMENT '사용 횟수',
    created_date DATETIME DEFAULT CURRENT_TIMESTAMP,

    INDEX idx_tag_name (tag_name),
    INDEX idx_use_count (use_count DESC)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='BBS 태그';

# 게시글-태그 연결 테이블
CREATE TABLE bbs_post_tags (
    post_id BIGINT NOT NULL,
    tag_id BIGINT NOT NULL,
    created_date DATETIME DEFAULT CURRENT_TIMESTAMP,

    PRIMARY KEY (post_id, tag_id),
    FOREIGN KEY (post_id) REFERENCES bbs_posts(post_id) ON DELETE CASCADE,
    FOREIGN KEY (tag_id) REFERENCES bbs_tags(tag_id) ON DELETE CASCADE,

    INDEX idx_tag_id (tag_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='게시글-태그 연결';

# 사용자 게시판 설정 테이블
CREATE TABLE bbs_user_settings (
    user_id BIGINT PRIMARY KEY,
    posts_per_page INT DEFAULT 20 COMMENT '페이지당 게시글 수',
    default_category_id BIGINT COMMENT '기본 카테고리',
    email_notification BOOLEAN DEFAULT FALSE COMMENT '이메일 알림 여부',
    comment_notification BOOLEAN DEFAULT TRUE COMMENT '댓글 알림 여부',
    hide_deleted_posts BOOLEAN DEFAULT TRUE COMMENT '삭제된 게시글 숨김',
    theme VARCHAR(20) DEFAULT 'DEFAULT' COMMENT '테마',
    created_date DATETIME DEFAULT CURRENT_TIMESTAMP,
    updated_date DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='사용자 게시판 설정';

# 신고 테이블
CREATE TABLE bbs_reports (
    report_id BIGINT AUTO_INCREMENT PRIMARY KEY,
    target_type VARCHAR(20) NOT NULL COMMENT '신고 대상 (POST, COMMENT)',
    target_id BIGINT NOT NULL COMMENT '신고 대상 ID',
    reporter_id BIGINT COMMENT '신고자 ID',
    reporter_ip VARCHAR(45) COMMENT '신고자 IP',
    reason_code VARCHAR(20) NOT NULL COMMENT '신고 사유 코드',
    reason_detail TEXT COMMENT '신고 상세 사유',
    status ENUM('PENDING', 'PROCESSED', 'REJECTED') DEFAULT 'PENDING' COMMENT '처리 상태 (PENDING, PROCESSED, REJECTED)',
    admin_memo TEXT COMMENT '관리자 메모',
    processed_by BIGINT COMMENT '처리한 관리자 ID',
    processed_date DATETIME COMMENT '처리 일시',
    created_date DATETIME DEFAULT CURRENT_TIMESTAMP,

    INDEX idx_target (target_type, target_id),
    INDEX idx_reporter (reporter_id),
    INDEX idx_status (status),
    INDEX idx_created_date (created_date)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='신고 관리';

# 게시판 통계 테이블 (일별 집계)
CREATE TABLE bbs_statistics (
    stat_id BIGINT AUTO_INCREMENT PRIMARY KEY,
    stat_date DATE NOT NULL COMMENT '통계 날짜',
    category_id BIGINT COMMENT '카테고리 ID (NULL이면 전체)',
    post_count INT DEFAULT 0 COMMENT '게시글 수',
    comment_count INT DEFAULT 0 COMMENT '댓글 수',
    view_count BIGINT DEFAULT 0 COMMENT '조회수',
    new_user_count INT DEFAULT 0 COMMENT '신규 사용자 수',
    active_user_count INT DEFAULT 0 COMMENT '활성 사용자 수',
    created_date DATETIME DEFAULT CURRENT_TIMESTAMP,

    UNIQUE KEY uk_date_category (stat_date, category_id),
    INDEX idx_stat_date (stat_date),
    INDEX idx_category_date (category_id, stat_date)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='게시판 통계';
